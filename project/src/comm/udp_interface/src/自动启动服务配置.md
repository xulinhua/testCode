#在x86主板上创建的systemd服务
sudo nano /etc/systemd/system/udp-server.service
填入以下内容：
[Unit]
Description=UDP Server for ROS2
After=network.target
StartLimitIntervalSec=0

[Service]
Type=simple
User=root
ExecStartPre=/bin/sleep 4
ExecStart=/bin/bash -c "source /opt/ros/humble/setup.bash && source /media/hs/Code/code/udp_interface/udp_interface/install/setup.bash && ros2 launch udp_server udp_server.launch.py"
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target

#在Jetson主板上创建的systemd服务
sudo nano /etc/systemd/system/udp-client.service
填入以下内容：
[Unit]
Description=UDP Client for ROS2
After=network.target
StartLimitIntervalSec=2

[Service]
Type=simple
User=user  # 改为user用户
Group=user # 添加用户组
Environment="HOME=/home/user"  # 设置HOME变量
Environment="ROS_DOMAIN_ID=0"  # 根据需要设置DOMAIN ID，与手动启动时一致
ExecStartPre=/bin/sleep 10  # 增加睡眠时间，确保系统准备就绪
ExecStart=/bin/bash -c "source /opt/ros/humble/setup.bash && source /home/user/project/install/setup.bash && ros2 launch udp_client udp_client.launch.py"
Restart=on-failure
RestartSec=5
StartLimitBurst=2  # 可选：设置最大重启次数，防止无限重启循环
TimeoutStopSec=15  # 设置停止超时为15秒
KillSignal=SIGINT  # 首先发送SIGINT信号（Ctrl+C）
FinalKillSignal=SIGKILL  # 超时后发送SIGKILL
SendSIGKILL=yes

[Install]
WantedBy=multi-user.target


#启用系统服务
在 x86 主板上执行:
sudo systemctl enable udp-server.service
sudo systemctl start udp-server.service

在 Jetson 主板上执行:
sudo systemctl enable udp-client.service
sudo systemctl start udp-client.service

#停止系统服务
在 x86 主板上执行:
sudo systemctl stop udp-server.service
或者通过ros2命令停止节点
ros2 node list
ros2 node info /udp_server
ros2 lifecycle set /udp_server shutdown
或者直接杀死进程
pkill -f 'udp_server_node'

在 Jetson 主板上执行:
sudo systemctl stop udp-client.service

或者通过ros2命令停止节点
ros2 node list  # 查看节点列表
ros2 node info /udp_client  # 查看节点信息
ros2 lifecycle set /udp_client shutdown  # 优雅关闭节点
或者直接杀死进程
pkill -f 'udp_client_node'

#禁用系统服务
在x86主板上执行：
sudo systemctl disable udp-server.service

在Jetson主板上执行：
sudo systemctl disable udp-client.service

#重新启动服务（如果不希望完全停止，只是修改配置）
如果只是修改了配置文件，并希望重新启动服务以应用新的配置（即不自动启动项目），可以重启服务：

在x86主板上执行：
sudo systemctl restart udp-server.service

在Jetson主板上执行：
sudo systemctl restart udp-client.service
